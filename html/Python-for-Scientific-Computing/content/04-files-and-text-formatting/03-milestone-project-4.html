🏁 Milestone Project 4: Data Logger and Report Writer

In this milestone project, you will build a complete scientific data logging pipeline: 1. Acquisition / Simulation: Generate a time series of thermal cooling data according to Newton’s Law of Cooling with simulated sensor noise. 2. File Export: Format the data into clean, aligned columns and write it to a text data log (cooling_log.dat) with metadata headers. 3. File Ingestion: Read the data file back into memory using line parsing and NumPy. 4. Analysis & Visualization: Compute statistical summary metrics and render a diagnostic plot with error bounds, saving the resulting graph to disk.

|                  DATA LOGGER AND REPORT WRITING PIPELINE                 |

|         ▼                                                                |

Complete Project Code: thermal_logger.py

Create a new file in Spyder named thermal_logger.py and run it:

"""
thermal_logger.py
=================
Milestone Project 4: Generates data, writes a formatted log file,
reads it back, and plots a verification figure.
"""

import numpy as np
import matplotlib.pyplot as plt


def generate_and_log_data(filename="cooling_log.dat"):
    """
    Simulates thermal cooling: T(t) = T_env + (T0 - T_env) * exp(-r * t)
    and logs the formatted data to a file.
    """
    print("\n1. Generating thermal cooling dataset...")
    
    # Physics parameters
    t_env = 22.0     # Ambient temperature in Celsius
    t_0 = 95.0       # Initial temperature (boiling liquid)
    r = 0.045        # Cooling rate constant (1/s)
    
    # Time points from 0 to 120 seconds
    time_pts = np.linspace(0, 120, 41)  # Every 3 seconds
    
    # Theoretical temperature decay
    t_clean = t_env + (t_0 - t_env) * np.exp(-r * time_pts)
    
    # Add sensor noise (+/- 0.6 °C)
    rng = np.random.default_rng(seed=2026)
    sensor_noise = rng.normal(0.0, 0.6, size=len(time_pts))
    t_measured = t_clean + sensor_noise
    
    # Write formatted file
    print(f"2. Writing formatted log to '{filename}'...")
    with open(filename, "w") as f:
        # File Header and Metadata
        f.write("# =========================================================\n")
        f.write("# EXPERIMENTAL DATA LOG: Newton's Law of Cooling\n")
        f.write("# Environment: 22.0 C | Initial Temp: 95.0 C | Rate: 0.045 /s\n")
        f.write("# =========================================================\n")
        f.write(f"{'Index':<6} | {'Time (s)':>10} | {'Theory (C)':>12} | {'Sensor (C)':>12}\n")
        f.write("# " + "-" * 50 + "\n")
        
        # Data Rows
        for i, (t_val, clean_val, meas_val) in enumerate(zip(time_pts, t_clean, t_measured), start=1):
            f.write(f"{i:<6d} | {t_val:10.2f} | {clean_val:12.3f} | {meas_val:12.3f}\n")
            
    print(f">> File '{filename}' successfully written ({len(time_pts)} records).")


def read_and_plot_data(filename="cooling_log.dat"):
    """
    Reads the formatted text file back from disk and plots the results.
    """
    print(f"\n3. Reading data back from '{filename}'...")
    
    times = []
    theory = []
    sensor = []
    
    with open(filename, "r") as f:
        for line in f:
            # Skip comment and header lines starting with '#'
            if line.startswith("#") or "Time" in line or not line.strip():
                continue
            
            # Split row by the '|' delimiter
            parts = line.strip().split("|")
            if len(parts) == 4:
                idx = int(parts[0])
                t_val = float(parts[1])
                theo_val = float(parts[2])
                sens_val = float(parts[3])
                
                times.append(t_val)
                theory.append(theo_val)
                sensor.append(sens_val)
                
    times = np.array(times)
    theory = np.array(theory)
    sensor = np.array(sensor)
    
    print(f"Parsed {len(times)} records. Generating report plot...")
    
    # 4. Plot and Save
    fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(8, 7), sharex=True, 
                                   gridspec_kw={'height_ratios': [3, 1]})
    
    # Top Panel: Cooling Curves
    ax1.plot(times, theory, "b-", linewidth=2, label="Theoretical Model")
    ax1.scatter(times, sensor, color="red", marker="o", s=30, label="Thermocouple Sensor", zorder=3)
    ax1.axhline(22.0, color="gray", linestyle="--", label="Ambient Temp (22 °C)")
    ax1.set(title="Liquid Thermal Cooling Profile", ylabel="Temperature (°C)")
    ax1.grid(True, linestyle=":", alpha=0.6)
    ax1.legend()
    
    # Bottom Panel: Residual Sensor Error
    residuals = sensor - theory
    ax2.axhline(0, color="black", linestyle="--", linewidth=1)
    ax2.stem(times, residuals, linefmt="k:", markerfmt="ro", basefmt="k-")
    ax2.set(title="Sensor Residual Errors", xlabel="Time (seconds)", ylabel="Error (°C)", ylim=(-2.5, 2.5))
    ax2.grid(True, linestyle=":", alpha=0.6)
    
    fig.tight_layout()
    plt.savefig("cooling_report.png", dpi=300, bbox_inches="tight")
    print(">> Report plot saved to 'cooling_report.png'!")
    plt.show()


if __name__ == "__main__":
    generate_and_log_data()
    read_and_plot_data()

When you run this script: 1. cooling_log.dat is created on disk with formatted headers and columns. 2. The file is opened, parsed, and checked for accuracy. 3. cooling_report.png is generated, showing the thermal decay curve and the residual sensor noise below it.
