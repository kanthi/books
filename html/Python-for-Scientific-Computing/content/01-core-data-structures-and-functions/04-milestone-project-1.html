1. Module Source Code: sci_toolkit.py

Save this file in your working folder as sci_toolkit.py:

"""
sci_toolkit.py
==============
A collection of scientific computing utilities for statistics, 
matrix transformations, and recursive algorithms.
"""

import math


# --- Part A: Statistical Helpers ---

def mean(data):
    """
    Compute the arithmetic mean of a numerical list.
    
    Parameters
    ----------
    data : list of float or int
        Sequence of numbers.
        
    Returns
    -------
    float
        The average value.
    """
    if not data:
        return 0.0
    return sum(data) / len(data)


def median(data):
    """
    Compute the median (middle value) of a list of numbers.
    """
    if not data:
        return 0.0
    sorted_data = sorted(data)
    n = len(sorted_data)
    mid = n // 2
    if n % 2 == 1:
        return float(sorted_data[mid])
    else:
        return (sorted_data[mid - 1] + sorted_data[mid]) / 2.0


def variance(data):
    """
    Calculate sample variance: sum((x - mean)^2) / (N - 1).
    """
    if len(data) < 2:
        return 0.0
    m = mean(data)
    return sum((x - m) ** 2 for x in data) / (len(data) - 1)


# --- Part B: Matrix Helpers (Lists of Lists) ---

def matrix_dimensions(matrix):
    """Return (rows, cols) of a 2D matrix."""
    if not matrix or not matrix[0]:
        return (0, 0)
    return (len(matrix), len(matrix[0]))


def row_totals(matrix):
    """Calculate the sum of each row in a 2D list."""
    return [sum(row) for row in matrix]


def column_totals(matrix):
    """Calculate the sum of each column in a 2D list."""
    if not matrix or not matrix[0]:
        return []
    num_cols = len(matrix[0])
    return [sum(matrix[r][c] for r in range(len(matrix))) for c in range(num_cols)]


def transpose(matrix):
    """
    Transpose a 2D matrix (swap rows and columns).
    Uses a clean nested list comprehension.
    """
    if not matrix or not matrix[0]:
        return []
    num_rows = len(matrix)
    num_cols = len(matrix[0])
    return [[matrix[r][c] for r in range(num_rows)] for c in range(num_cols)]


# --- Part C: Recursive Helpers ---

def recursive_power(base, exponent):
    """Compute base^exponent recursively for non-negative integers."""
    if exponent == 0:
        return 1.0
    elif exponent < 0:
        return 1.0 / recursive_power(base, -exponent)
    return base * recursive_power(base, exponent - 1)


def recursive_gcd(a, b):
    """Euclidean algorithm for greatest common divisor: gcd(a, b)."""
    if b == 0:
        return a
    return recursive_gcd(b, a % b)


# --- Self-Testing Guard ---
if __name__ == "__main__":
    print("Running internal self-check for sci_toolkit.py...")
    test_data = [10.0, 20.0, 30.0, 40.0, 50.0]
    print("Internal Mean test:", mean(test_data), "(Expected: 30.0)")

2. Test Driver Script: run_toolkit_tests.py

Save this second file in the same directory and execute it with F5 in Spyder:

"""
run_toolkit_tests.py
====================
Main program that imports and verifies sci_toolkit.
"""

import sci_toolkit as tk

print("=" * 60)
print("     SCIENTIFIC UTILITY TOOLKIT VERIFICATION SUITE")
print("=" * 60)

# 1. Testing Statistics
sensor_readings = [23.4, 25.1, 22.8, 24.6, 23.9, 24.2, 25.0]
print("\n--- 1. Statistics on Sensor Readings ---")
print("Dataset:      ", sensor_readings)
print(f"Mean:          {tk.mean(sensor_readings):.3f}")
print(f"Median:        {tk.median(sensor_readings):.3f}")
print(f"Variance:      {tk.variance(sensor_readings):.4f}")

# 2. Testing 2D Matrices
sensor_grid = [
    [12, 15, 18],
    [20, 22, 24],
    [30, 31, 35],
    [40, 42, 44]
]
rows, cols = tk.matrix_dimensions(sensor_grid)
print(f"\n--- 2. Matrix Utilities (Grid: {rows}x{cols}) ---")
print("Original Matrix:")
for row in sensor_grid:
    print("  ", row)

print("Row sums:      ", tk.row_totals(sensor_grid))
print("Column sums:   ", tk.column_totals(sensor_grid))

grid_t = tk.transpose(sensor_grid)
t_rows, t_cols = tk.matrix_dimensions(grid_t)
print(f"Transposed Matrix ({t_rows}x{t_cols}):")
for row in grid_t:
    print("  ", row)

# 3. Testing Recursive Helpers
print("\n--- 3. Recursive Functions ---")
base, exp = 3.0, 4
print(f"{base}^{exp} = {tk.recursive_power(base, exp)} (Expected: 81.0)")

num1, num2 = 48, 18
print(f"GCD({num1}, {num2}) = {tk.recursive_gcd(num1, num2)} (Expected: 6)")

print("\n" + "=" * 60)
print("     ALL TOOLKIT CHECKS COMPLETED SUCCESSFULLY")
print("=" * 60)
