|                  CAPSTONE: ROBUST DATA ANALYSIS PIPELINE                 |

|  [ 1. Ingestion Engine ]                                                 |
|     - Reads 'spectrometry_raw.csv' with corrupted entries & comments     |
|     - Uses try-except guards to filter bad data without crashing         |

|  [ 2. Numerical Processing (NumPy) ]                                     |

1. Pipeline Architecture

The pipeline consists of four modular functions designed for clean separation of concerns: 1. generate_mock_laboratory_file(filename) — Generates a realistic CSV file with headers, comments, valid sensor data, and corrupted values. 2. load_and_clean_data(filename) — Opens the file, applies defensive exception handling to discard corrupted rows, and returns clean NumPy arrays. 3. analyze_spectrum(wavelengths, intensities) — Performs numerical baseline removal, smooths signal noise, locates emission peaks, and calculates total energy. 4. generate_visual_and_text_report(...) — Renders the multi-panel figure and writes a formatted executive summary file.

2. Complete Pipeline Implementation: pipeline_capstone.py

Create a file named pipeline_capstone.py in your Spyder editor and run it (F5):

"""
pipeline_capstone.py
====================
Capstone Project: An end-to-end robust scientific analysis pipeline
incorporating File I/O, Exception Handling, NumPy, and Matplotlib.
"""

import os
import numpy as np
import matplotlib.pyplot as plt


def generate_mock_laboratory_file(filename="spectrometry_raw.csv"):
    """Generates mock optical emission spectrometry data with corrupted rows."""
    print(f"[Stage 1] Generating raw laboratory file: '{filename}'...")
    
    # 200 wavelength points from 400 nm to 700 nm (visible light spectrum)
    wavelengths = np.linspace(400.0, 700.0, 201)
    
    # Emission peaks at 480nm (Hydrogen-beta) and 656nm (Hydrogen-alpha)
    peak1 = 35.0 * np.exp(-0.5 * ((wavelengths - 486.1) / 8.0)**2)
    peak2 = 80.0 * np.exp(-0.5 * ((wavelengths - 656.3) / 10.0)**2)
    baseline = 15.0 + 0.02 * wavelengths
    
    rng = np.random.default_rng(seed=2026)
    noise = rng.normal(0, 1.5, size=len(wavelengths))
    intensity = baseline + peak1 + peak2 + noise
    
    with open(filename, "w") as f:
        f.write("# OPTICAL EMISSION SPECTROMETER LAB RUN 42\n")
        f.write("# Instrument: OceanOptics-HR2000 | Operator: College Lab Team\n")
        f.write("Wavelength(nm),Intensity(counts)\n")
        
        for i, (w, intens) in enumerate(zip(wavelengths, intensity)):
            # Introduce intentional realistic data corruption at isolated indices
            if i == 45:
                f.write("467.50,SENSOR_COMM_TIMEOUT\n")  # Corrupted float
            elif i == 110:
                f.write("CORRUPTED_LINE_DROPPED_SYNC\n")   # Missing column
            elif i == 160:
                f.write("640.00,-9999.00\n")             # Out-of-bounds error code
            else:
                f.write(f"{w:.2f},{intens:.3f}\n")
                
    print(f"         Generated {len(wavelengths)} rows with 3 intentional flaws.")


def load_and_clean_data(filename="spectrometry_raw.csv"):
    """
    Reads CSV file with defensive exception handling, skipping bad records.
    """
    print(f"[Stage 2] Ingesting and sanitizing data from '{filename}'...")
    
    clean_w = []
    clean_i = []
    dropped_count = 0
    
    if not os.path.exists(filename):
        raise FileNotFoundError(f"Required data file '{filename}' was not found.")
        
    with open(filename, "r") as f:
        for line_num, line in enumerate(f, start=1):
            line_str = line.strip()
            
            # Skip comments and header
            if line_str.startswith("#") or "Wavelength" in line_str or not line_str:
                continue
                
            parts = line_str.split(",")
            try:
                if len(parts) != 2:
                    raise IndexError(f"Expected 2 columns, found {len(parts)}")
                    
                w_val = float(parts[0])
                i_val = float(parts[1])
                
                # Check physical validity (negative readings indicate hardware error)
                if i_val < 0 or w_val <= 0:
                    raise ValueError(
                        f"Physically invalid measurement: ({w_val}, {i_val})"
                    )
                    
                clean_w.append(w_val)
                clean_i.append(i_val)
                
            except (ValueError, IndexError) as err:
                dropped_count += 1
                # In production code, log the error rather than crashing
                # print(f"         Skipping row {line_num}: {err}")
                
    print(
        f"         Successfully parsed {len(clean_w)} valid rows "
        f"(Filtered out {dropped_count} bad rows)."
    )
    return np.array(clean_w), np.array(clean_i)


def analyze_spectrum(wavelengths, intensities):
    """Computes spectral features, peak wavelengths, and background."""
    print("[Stage 3] Running NumPy numerical processing...")
    
    # 1. Baseline estimation using linear fit on edges
    left_mask = wavelengths < 430.0
    right_mask = wavelengths > 680.0
    edge_w = np.concatenate([wavelengths[left_mask], wavelengths[right_mask]])
    edge_i = np.concatenate([intensities[left_mask], intensities[right_mask]])
    
    # Fit line: y = m*x + b
    slope, intercept = np.polyfit(edge_w, edge_i, deg=1)
    baseline = slope * wavelengths + intercept
    
    # 2. Baseline subtraction
    corrected_intensity = intensities - baseline
    
    # 3. Peak Identification
    peak_idx = np.argmax(corrected_intensity)
    peak_w = wavelengths[peak_idx]
    peak_max_counts = corrected_intensity[peak_idx]
    
    # 4. Integrated Signal Area (Trapezoidal Rule)
    total_area = np.trapezoid(np.maximum(corrected_intensity, 0), wavelengths)
    
    results = {
        "baseline": baseline,
        "corrected": corrected_intensity,
        "peak_wavelength": peak_w,
        "peak_intensity": peak_max_counts,
        "integrated_area": total_area,
        "mean_noise_std": np.std(edge_i - (slope * edge_w + intercept))
    }
    return results


def generate_visual_and_text_report(w, i_raw, analysis, 
                                    fig_name="spectrometry_analysis.png", 
                                    report_name="spectrometry_report.txt"):
    """Creates a publication plot and formats an executive summary."""
    print("[Stage 4] Producing output visualizations and executive report...")
    
    # --- Part A: 3-Panel Matplotlib Figure ---
    fig, (ax1, ax2, ax3) = plt.subplots(3, 1, figsize=(9, 10), sharex=True)
    fig.suptitle(
        "Spectrometry Acquisition & Diagnostic Analysis",
        fontsize=14, fontweight="bold",
    )
    
    # Subplot 1: Raw Signal and Baseline
    ax1.plot(w, i_raw, "b.", markersize=3, label="Raw Sensor Readings")
    ax1.plot(
        w, analysis["baseline"], "r--", linewidth=1.8,
        label="Estimated Linear Baseline",
    )
    ax1.set(ylabel="Intensity (counts)", title="Raw Spectral Profile & Baseline")
    ax1.legend(loc="upper left")
    ax1.grid(True, linestyle=":", alpha=0.6)
    
    # Subplot 2: Corrected Signal & Peak Marker
    corr = analysis["corrected"]
    ax2.plot(w, corr, "darkgreen", linewidth=1.5, label="Baseline-Corrected Signal")
    ax2.plot(
        analysis["peak_wavelength"],
        analysis["peak_intensity"],
        "r*",
        markersize=14,
        label=f"Dominant Peak: {analysis['peak_wavelength']:.1f} nm",
    )
    ax2.fill_between(
        w, 0, np.maximum(corr, 0),
        color="green", alpha=0.15, label="Integrated Band",
    )
    ax2.set(ylabel="Corrected (counts)", title="Baseline-Corrected Emission Peaks")
    ax2.legend(loc="upper left")
    ax2.grid(True, linestyle=":", alpha=0.6)
    
    # Subplot 3: Residual Edge Noise
    residuals = i_raw - analysis["baseline"]
    ax3.axhline(0, color="black", linestyle="--")
    ax3.plot(w, residuals, color="gray", linewidth=1.0, label="Noise Residuals")
    ax3.set(xlabel="Wavelength (nm)", ylabel="Residuals", title="Sensor Noise Floor")
    ax3.legend(loc="upper left")
    ax3.grid(True, linestyle=":", alpha=0.6)
    
    fig.tight_layout()
    plt.savefig(fig_name, dpi=300, bbox_inches="tight")
    print(f"         Saved figure: '{fig_name}'")
    plt.show()
    
    # --- Part B: Executive Text Summary ---
    with open(report_name, "w") as f:
        f.write("=" * 65 + "\n")
        f.write("      EXECUTIVE SPECTROMETRY ANALYSIS SUMMARY REPORT\n")
        f.write("=" * 65 + "\n")
        f.write(f"Wavelength Range Monitored:   [{w[0]:.1f} nm - {w[-1]:.1f} nm]\n")
        f.write(f"Valid Samples Processed:      {len(w):d}\n")
        f.write(
            "Estimated Noise Floor (Std):  "
            f"{analysis['mean_noise_std']:.3f} counts\n"
        )
        f.write("-" * 65 + "\n")
        f.write("IDENTIFIED SPECTRAL FEATURES:\n")
        f.write(
            "  * Dominant Emission Peak:   "
            f"{analysis['peak_wavelength']:8.2f} nm\n"
        )
        f.write(
            "  * Peak Signal Height:       "
            f"{analysis['peak_intensity']:8.2f} counts\n"
        )
        f.write(
            "  * Total Integrated Power:   "
            f"{analysis['integrated_area']:8.2f} counts*nm\n"
        )
        f.write("=" * 65 + "\n")
        f.write("STATUS: Pipeline executed with ZERO unhandled exceptions.\n")
        
    print(f"         Saved summary: '{report_name}'")


if __name__ == "__main__":
    # Execute full pipeline end-to-end
    raw_file = "spectrometry_raw.csv"
    generate_mock_laboratory_file(raw_file)
    w_clean, i_clean = load_and_clean_data(raw_file)
    spectral_metrics = analyze_spectrum(w_clean, i_clean)
    generate_visual_and_text_report(w_clean, i_clean, spectral_metrics)
    print("\n[SUCCESS] Capstone analysis pipeline completed successfully!")
